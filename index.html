function getUserLocation() { // التأكد من أن المتصفح يدعم خاصية تحديد الموقع if (!navigator.geolocation) { alert("عذراً، متصفحك لا يدعم خاصية تحديد الموقع."); return; } // خيارات الدقة العالية const options = { enableHighAccuracy: true, // تفعيل الدقة العالية (GPS) timeout: 5000, // مدة الانتظار 5 ثوانٍ maximumAge: 0 // عدم استخدام موقع قديم مخزن }; navigator.geolocation.getCurrentPosition(success, error, options); function success(position) { const latitude = position.coords.latitude; const longitude = position.coords.longitude; // تحديث قيم الحقول في الواجهة تلقائياً document.getElementById('lat').value = latitude.toFixed(6); document.getElementById('lon').value = longitude.toFixed(6); console.log(`الموقع الحالي: ${latitude}, ${longitude}`); alert("تم تحديد موقعك بنجاح!"); } function error(err) { switch(err.code) { case err.PERMISSION_DENIED: alert("يرجى السماح للتطبيق بالوصول إلى الموقع."); break; case err.POSITION_UNAVAILABLE: alert("معلومات الموقع غير متوفرة حالياً."); break; case err.TIMEOUT: alert("انتهت مهلة طلب الموقع، حاول مرة أخرى."); break; default: alert("حدث خطأ غير معروف في تحديد الموقع."); } } }
